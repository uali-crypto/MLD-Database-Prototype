<script>

  const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbza1KNMLZGFDW1eo149emlLSjLr4wwhBRMZaGrzFiKOwEb8pA3MZpOMRg9H7nMPn674dg/exec";


  // ORCID Lookup

  document.getElementById("orcid-lookup-btn").addEventListener("click", async () => {

    const orcidInput = document.getElementById("orcid").value.trim();

    const resultEl = document.getElementById("orcid-result");


    if (!orcidInput) {

      resultEl.textContent = "Please enter an ORCID iD.";

      resultEl.className = "orcid-result error";

      return;

    }


    const orcidClean = orcidInput.replace("https://orcid.org/", "");


    try {

      const response = await fetch(`https://pub.orcid.org/v3.0/${orcidClean}/person`, {

        headers: { "Accept": "application/json" }

      });


      if (!response.ok) {

        resultEl.textContent = "ORCID iD not found. Please check and try again.";

        resultEl.className = "orcid-result error";

        return;

      }


      const data = await response.json();

      const given = data.name?.["given-names"]?.value || "";

      const family = data.name?.["family-name"]?.value || "";

      const fullName = `${given} ${family}`.trim();


      if (fullName) {

        resultEl.textContent = `✓ Found: ${fullName}`;

        resultEl.className = "orcid-result success";

      } else {

        resultEl.textContent = "Name not publicly available for this ORCID iD.";

        resultEl.className = "orcid-result error";

      }


    } catch (err) {

      resultEl.textContent = "Error contacting ORCID. Please try again.";

      resultEl.className = "orcid-result error";

    }

  });


  // Form Submission

  document.getElementById("mld-form").addEventListener("submit", async (e) => {

    e.preventDefault();


    const statusEl = document.getElementById("form-status");

    const orcidResult = document.getElementById("orcid-result").textContent;


    if (!orcidResult.startsWith("✓")) {

      statusEl.textContent = "Please look up and verify your ORCID iD before submitting.";

      statusEl.className = "form-status error";

      return;

    }


    const payload = {

      orcid: document.getElementById("orcid").value.trim(),

      contributorName: orcidResult.replace("✓ Found: ", ""),

      doi: document.getElementById("doi").value.trim(),

      precursor1: document.getElementById("precursor1").value.trim(),

      precursor2: document.getElementById("precursor2").value.trim(),

      temp: document.getElementById("temp").value.trim(),

    };


    statusEl.textContent = "Submitting...";

    statusEl.className = "form-status";


    try {

      await fetch(APPS_SCRIPT_URL, {

        method: "POST",

        mode: "no-cors",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify(payload),

      });


      statusEl.textContent = "✓ Entry submitted successfully. Thank you!";

      statusEl.className = "form-status success";

      document.getElementById("mld-form").reset();

      document.getElementById("orcid-result").textContent = "";


    } catch (err) {

      statusEl.textContent = "Submission failed. Please try again.";

      statusEl.className = "form-status error";

    }

  });

</script>
